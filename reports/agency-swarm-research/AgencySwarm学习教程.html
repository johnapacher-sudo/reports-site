二、快速开始

2.1 Hello World

创建最简单的智能体系统：

# hello_world.py
import asyncio
from agency_swarm import Agent, Agency

async def main():
    # 创建一个智能体
    assistant = Agent(
        name="Assistant",
        description="一个友好的助手",
        instructions="你是一个乐于助人的助手，简洁地回答问题。"
    )

    # 创建 Agency（单个智能体也需要 Agency）
    agency = Agency(assistant)

    # 发送消息
    response = await agency.get_response("你好！请介绍一下你自己。")
    print(response)

if __name__ == "__main__":
    asyncio.run(main())

运行：

python hello_world.py

2.2 第一个多智能体示例

创建一个简单的双智能体协作系统：

# first_agency.py
import asyncio
from agency_swarm import Agent, Agency

async def main():
    # 创建经理智能体
    manager = Agent(
        name="Manager",
        description="项目经理，负责任务分配和协调",
        instructions="""
        你是一位经验丰富的项目经理。
        - 分析用户请求
        - 将任务分配给合适的团队成员
        - 汇总结果返回给用户
        """
    )

    # 创建开发者智能体
    developer = Agent(
        name="Developer",
        description="软件工程师，负责代码实现",
        instructions="""
        你是一位全栈开发者。
        - 根据需求编写代码
        - 提供清晰的技术说明
        - 确保代码质量
        """
    )

    # 创建 Agency，定义通信流
    agency = Agency(
        manager,  # 入口智能体
        communication_flows=[
            manager > developer  # Manager 可以发给 Developer
        ]
    )

    # 执行任务
    response = await agency.get_response(
        "帮我写一个 Python 函数，计算斐波那契数列的第 n 项"
    )
    print(response)

if __name__ == "__main__":
    asyncio.run(main())

2.3 理解输出

运行后你会看到类似输出：

Manager 正在处理请求...
Manager 决定委托给 Developer...
Developer 正在执行任务...
Developer 完成任务，返回结果...
Manager 汇总结果...

最终结果:
这是一个计算斐波那契数列的 Python 函数：

def fibonacci(n):
    if n <= 1:
        return n
    return fibonacci(n-1) + fibonacci(n-2)
...

3.2 核心概念详解

Agent（智能体）

Agent 是 Agency Swarm 的基本构建单元。

每个 Agent 包含：
├── name          - 唯一标识符
├── description   - 角色描述（用于智能体间了解彼此）
├── instructions  - 详细指令（决定智能体行为）
├── tools         - 可用工具列表
├── model         - 使用的 LLM 模型
└── model_settings - 模型参数配置

Agency（编排器）

Agency 管理智能体间的协作。

职责：
├── 接收用户请求
├── 路由消息到正确的智能体
├── 管理通信流
├── 维护会话状态
└── 返回最终结果

Communication Flows（通信流）

定义智能体间的通信方向。

使用 > 运算符：
  ceo > dev   表示 CEO 可以发送消息给 Developer

特点：
├── 单向通信
├── 必须显式定义
└── 支持多对多关系

Tool（工具）

工具是智能体与外部世界交互的方式。

特点：
├── 使用 @function_tool 装饰器定义
├── 基于 Pydantic 自动验证参数
├── 自动生成 JSON Schema
└── 支持复杂类型

3.3 术语表

术语 英文 定义
智能体 Agent 具有特定角色和能力的 AI 实体
编排器 Agency 管理智能体协作的协调者
通信流 Communication Flow 智能体间消息传递的方向定义
工具 Tool 智能体可调用的函数
指令 Instructions 定义智能体行为的文本
共享指令 Shared Instructions 所有智能体共享的全局指令
线程 Thread 一次对话会话
上下文 Context 运行时的状态和环境信息

文件操作工具

import os

@function_tool
def read_file(filepath: str) -> str:
    """
    读取文件内容。

    Args:
        filepath: 文件路径

    Returns:
        文件内容
    """
    if not os.path.exists(filepath):
        return f"错误：文件 {filepath} 不存在"

    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()

@function_tool
def write_file(filepath: str, content: str) -> str:
    """
    写入文件。

    Args:
        filepath: 文件路径
        content: 要写入的内容

    Returns:
        操作结果
    """
    try:
        with open(filepath, 'w', encoding='utf-8') as f:
            f.write(content)
        return f"成功写入文件 {filepath}"
    except Exception as e:
        return f"错误：{str(e)}"

4.5 状态持久化

基础持久化

import json

def load_threads_callback():
    """加载保存的线程"""
    try:
        with open('threads.json', 'r') as f:
            return json.load(f)
    except FileNotFoundError:
        return {}

def save_threads_callback(threads):
    """保存线程状态"""
    with open('threads.json', 'w') as f:
        json.dump(threads, f, indent=2, ensure_ascii=False)

agency = Agency(
    ceo,
    communication_flows=[ceo > dev],
    load_threads_callback=load_threads_callback,
    save_threads_callback=save_threads_callback
)

使用数据库持久化

import sqlite3
import json

class ThreadPersistence:
    def __init__(self, db_path='threads.db'):
        self.conn = sqlite3.connect(db_path)
        self._init_db()

    def _init_db(self):
        self.conn.execute('''
            CREATE TABLE IF NOT EXISTS threads (
                thread_id TEXT PRIMARY KEY,
                data TEXT,
                updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')

    def load(self):
        cursor = self.conn.execute('SELECT thread_id, data FROM threads')
        return {row[0]: json.loads(row[1]) for row in cursor}

    def save(self, threads):
        for thread_id, data in threads.items():
            self.conn.execute('''
                INSERT OR REPLACE INTO threads (thread_id, data)
                VALUES (?, ?)
            ''', (thread_id, json.dumps(data)))
        self.conn.commit()

# 使用
persistence = ThreadPersistence()

agency = Agency(
    ceo,
    communication_flows=[ceo > dev],
    load_threads_callback=persistence.load,
    save_threads_callback=persistence.save
)

5.2 错误处理与重试

import asyncio
from tenacity import retry, stop_after_attempt, wait_exponential

@retry(
    stop=stop_after_attempt(3),
    wait=wait_exponential(multiplier=1, min=2, max=10)
)
async def run_with_retry(agency, message):
    """带重试的任务执行"""
    try:
        response = await agency.get_response(message)
        return response
    except Exception as e:
        print(f"执行失败: {e}")
        raise

async def main():
    agency = Agency(ceo, communication_flows=[ceo > dev])

    try:
        response = await run_with_retry(agency, "创建一个项目")
        print(response)
    except Exception as e:
        print(f"所有重试都失败: {e}")

5.3 性能优化

并行执行

import asyncio

async def parallel_tasks():
    agency = Agency(ceo, communication_flows=[ceo > dev])

    # 并行执行多个任务
    tasks = [
        agency.get_response("任务1"),
        agency.get_response("任务2"),
        agency.get_response("任务3"),
    ]

    results = await asyncio.gather(*tasks)
    return results

缓存策略

from functools import lru_cache
import hashlib

class CachedAgency:
    def __init__(self, agency):
        self.agency = agency
        self.cache = {}

    def _get_cache_key(self, message):
        return hashlib.md5(message.encode()).hexdigest()

    async def get_response(self, message):
        key = self._get_cache_key(message)

        if key in self.cache:
            print("使用缓存结果")
            return self.cache[key]

        response = await self.agency.get_response(message)
        self.cache[key] = response
        return response

FastAPI 部署

# api.py
from fastapi import FastAPI
from pydantic import BaseModel
from agency_swarm import Agent, Agency
import asyncio

app = FastAPI()

# 初始化 Agency
ceo = Agent(name="CEO", ...)
dev = Agent(name="Developer", ...)
agency = Agency(ceo, communication_flows=[ceo > dev])

class MessageRequest(BaseModel):
    message: str
    user_id: str = None

@app.post("/chat")
async def chat(request: MessageRequest):
    response = await agency.get_response(request.message)
    return {"response": response}

@app.post("/chat/stream")
async def chat_stream(request: MessageRequest):
    async def generate():
        async for chunk in agency.get_response_stream(request.message):
            yield f"data: {chunk}\n\n"

    return StreamingResponse(
        generate(),
        media_type="text/event-stream"
    )

# code_review_agency.py
import asyncio
from agency_swarm import Agent, Agency, function_tool

# 定义工具
@function_tool
def read_github_pr(repo: str, pr_number: int) -> str:
    """读取 GitHub PR 的代码变更"""
    # 实际实现会调用 GitHub API
    return f"PR #{pr_number} in {repo} 的代码变更..."

@function_tool
def post_review_comment(pr_number: int, comment: str) -> str:
    """在 PR 上发布审查评论"""
    return f"已发布评论到 PR #{pr_number}"

# 定义智能体
architect = Agent(
    name="Architect",
    description="系统架构师，负责审查架构和设计",
    instructions="""
    你是一位资深架构师。

    审查重点：
    1. 代码架构是否合理
    2. 模块划分是否清晰
    3. 是否遵循设计模式
    4. 可扩展性如何

    输出格式：
    - 架构评估
    - 改进建议
    - 风险提示
    """,
    tools=[read_github_pr]
)

security_expert = Agent(
    name="SecurityExpert",
    description="安全专家，负责审查安全漏洞",
    instructions="""
    你是一位安全专家。

    审查重点：
    1. SQL 注入风险
    2. XSS 漏洞
    3. 敏感信息泄露
    4. 权限控制
    5. 加密使用

    输出格式：
    - 安全评分
    - 漏洞列表
    - 修复建议
    """,
    tools=[read_github_pr]
)

reviewer = Agent(
    name="Reviewer",
    description="代码审查协调者",
    instructions="""
    你是代码审查的协调者。

    职责：
    1. 分配审查任务
    2. 汇总审查意见
    3. 做出最终决定
    """,
    tools=[post_review_comment]
)

# 创建 Agency
agency = Agency(
    reviewer,
    communication_flows=[
        reviewer > architect,
        reviewer > security_expert
    ]
)

async def review_pr(repo: str, pr_number: int):
    """审查 PR"""
    message = f"请审查 {repo} 的 PR #{pr_number}"
    response = await agency.get_response(message)
    return response

# 使用
if __name__ == "__main__":
    asyncio.run(review_pr("my-org/my-repo", 123))

    """,
    tools=[save_draft]
)

fact_checker = Agent(
    name="FactChecker",
    description="事实核查员",
    instructions="""
    你是一位严谨的事实核查员。

    核查要点：
    1. 数据准确性
    2. 引用来源
    3. 逻辑一致性
    4. 时效性
    """
)

6.3 案例三：智能客服系统

# customer_service_agency.py
from agency_swarm import Agent, Agency, function_tool
from typing import Optional

@function_tool
def query_order(order_id: str) -> str:
    """查询订单状态"""
    # 模拟数据库查询
    orders = {
        "12345": {"status": "已发货", "tracking": "SF123456"},
        "67890": {"status": "处理中", "tracking": None}
    }
    order = orders.get(order_id)
    if order:
        return f"订单状态: {order['status']}, 物流单号: {order['tracking'] or '暂无'}"
    return "未找到订单"

@function_tool
def query_faq(question: str) -> str:
    """查询常见问题"""
    faqs = {
        "退货": "收到商品后7天内可申请退货...",
        "配送": "一般情况下2-3个工作日送达...",
        "支付": "支持支付宝、微信、银行卡..."
    }
    for keyword, answer in faqs.items():
        if keyword in question:
            return answer
    return "未找到相关问题"

@function_tool
def create_ticket(
    user_id: str,
    issue: str,
    priority: str = "normal"
) -> str:
    """创建工单"""
    return f"已创建工单: 用户 {user_id}, 问题: {issue}, 优先级: {priority}"

# 定义智能体
triage = Agent(
    name="Triage",
    description="客服分流，负责问题分类和路由",
    instructions="""
    你是客服分流员。

    职责：
    1. 理解用户问题
    2. 判断问题类型
    3. 路由到合适的专员

    问题类型：
    - 订单问题 → OrderSpecialist
    - 产品咨询 → ProductExpert
    - 投诉建议 → Supervisor
    """,
    model="gpt-5-mini"
)

order_specialist = Agent(
    name="OrderSpecialist",
    description="订单专员，处理订单相关问题",
    instructions="""
    你是订单专员。

    可处理：
    - 订单查询
    - 物流跟踪
    - 订单修改
    - 退款处理
    """,
    tools=[query_order, create_ticket]
)

product_expert = Agent(
    name="ProductExpert",
    description="产品专家，回答产品相关问题",
    instructions="""
    你是产品专家。

    负责：
    - 产品功能介绍
    - 使用方法指导
    - 常见问题解答
    """,
    tools=[query_faq]
)

supervisor = Agent(
    name="Supervisor",
    description="主管，处理复杂问题和投诉",
    instructions="""
    你是客服主管。

    负责：
    - 处理投诉
    - 复杂问题升级
    - 特殊情况处理
    """,
    tools=[create_ticket]
)

# 创建 Agency
agency = Agency(
    triage,
    communication_flows=[
        triage > order_specialist,
        triage > product_expert,
        triage > supervisor
    ]
)

7.3 最佳实践问题

Q: 如何设计智能体职责？

原则：
1. 单一职责 - 每个智能体只做一件事
2. 清晰边界 - 避免职责重叠
3. 适度粒度 - 3-5 个智能体通常足够

示例：
✅ 好的设计
- Architect: 架构设计
- Developer: 代码实现
- Tester: 测试验证

❌ 避免
- SuperDev: 架构 + 开发 + 测试（职责过多）

Q: 如何调试智能体行为？

# 1. 使用详细日志
import logging
logging.basicConfig(level=logging.DEBUG)

# 2. 检查指令是否清晰
# 让智能体解释它理解的任务

# 3. 逐步测试
# 先测试单个智能体，再测试协作

# 4. 查看消息流
# 在 Agency 中添加日志回调
