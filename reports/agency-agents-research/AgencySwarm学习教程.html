三、核心概念

3.1 Agent

Agent 是 Agency Swarm 的基本执行单元。

from agency_swarm import Agent, ModelSettings

agent = Agent(
    name="MyAgent",
    description="Agent 角色描述",
    instructions="详细指令...",
    model="gpt-4o",
    model_settings=ModelSettings(
        temperature=0.7,
        max_tokens=4000,
    ),
)

Agent 核心属性：

属性 类型 说明
name str Agent 名称（唯一标识）
description str 角色描述
instructions str 系统指令/提示词
model str 模型名称
model_settings ModelSettings 模型配置
tools list 可用工具列表
files_folder Path 文件目录
tools_folder Path 工具目录
schemas_folder Path OpenAPI Schema 目录

3.2 Agency

Agency 是多个 Agent 的协作容器。

from agency_swarm import Agency

agency = Agency(
    entry_agent,                    # 入口 Agent
    communication_flows=[           # 通信流定义
        agent1 > agent2,
        agent1 > agent3,
    ],
    shared_instructions="...",      # 共享指令
)

3.3 Communication Flow（通信流）

通信流 定义了 Agent 之间如何通信。

# 使用 > 操作符定义通信方向
flow1 = ceo > developer  # CEO -> Developer
flow2 = ceo > va         # CEO -> Virtual Assistant
flow3 = dev > tester     # Developer -> Tester

# 链式通信流
chain = ceo > dev > tester  # CEO -> Developer -> Tester

五、工具系统

5.1 使用 @function_tool 装饰器（推荐）

from agency_swarm import function_tool

@function_tool
def search_web(query: str) -> str:
    """在网络上搜索信息。

    Args:
        query: 搜索查询字符串

    Returns:
        搜索结果
    """
    # 实现搜索逻辑
    return f"搜索 '{query}' 的结果..."

@function_tool
def read_file(file_path: str, encoding: str = "utf-8") -> str:
    """读取文件内容。

    Args:
        file_path: 文件路径
        encoding: 文件编码，默认 utf-8

    Returns:
        文件内容
    """
    with open(file_path, encoding=encoding) as f:
        return f.read()

# 使用工具
agent = Agent(
    name="Researcher",
    tools=[search_web, read_file],
)

5.2 使用 BaseTool 类

from agency_swarm.tools import BaseTool
from pydantic import Field

class DatabaseQuery(BaseTool):
    """执行数据库查询。"""

    query: str = Field(
        ...,
        description="SQL 查询语句"
    )
    database: str = Field(
        default="main",
        description="数据库名称"
    )

    def run(self):
        """执行查询。"""
        # 实现查询逻辑
        return f"执行查询: {self.query}"

# 使用工具
agent = Agent(
    name="DataAgent",
    tools=[DatabaseQuery],
)

from agency_swarm.tools import SendMessage
from pydantic import BaseModel, Field

class SendMessageWithContext(SendMessage):
    """带上下文跟踪的 send_message。"""

    class ExtraParams(BaseModel):
        key_moments: str = Field(
            description="记录关键时刻和决策点..."

七、状态持久化

7.1 使用 ThreadManager

from agency_swarm import Agency
import json

def load_messages() -> list[dict]:
    """从存储加载消息。"""
    try:
        with open("messages.json") as f:
            return json.load(f)
    except FileNotFoundError:
        return []

def save_messages(messages: list[dict]) -> None:
    """保存消息到存储。"""
    with open("messages.json", "w") as f:
        json.dump(messages, f, indent=2)

agency = Agency(
    ceo,
    communication_flows=[ceo > developer],
    load_threads_callback=load_messages,
    save_threads_callback=save_messages,
)

7.2 使用 PersistenceHooks

from agency_swarm.hooks import PersistenceHooks

hooks = PersistenceHooks(
    load_callback=load_messages,
    save_callback=save_messages,
)

agency = Agency(
    ceo,
    communication_flows=[ceo > developer],
    hooks=hooks,
)

7.3 数据库持久化

import sqlite3
import json

class DatabasePersistence:
    def __init__(self, db_path: str = "agency.db"):
        self.conn = sqlite3.connect(db_path)
        self._create_table()

    def _create_table(self):
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS messages (
                id INTEGER PRIMARY KEY,
                thread_id TEXT,
                agent TEXT,
                role TEXT,
                content TEXT,
                timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
            )
        """)

    def load(self, thread_id: str) -> list[dict]:
        cursor = self.conn.execute(
            "SELECT * FROM messages WHERE thread_id = ?",
            (thread_id,)
        )
        return [dict(row) for row in cursor.fetchall()]

    def save(self, thread_id: str, messages: list[dict]):
        for msg in messages:
            self.conn.execute(
                "INSERT INTO messages (thread_id, agent, role, content) VALUES (?, ?, ?, ?)",
                (thread_id, msg.get("agent"), msg.get("role"), json.dumps(msg.get("content")))
            )
        self.conn.commit()

# 使用
persistence = DatabasePersistence()

agency = Agency(
    ceo,
    communication_flows=[ceo > developer],
    load_threads_callback=lambda: persistence.load("main"),
    save_threads_callback=lambda msgs: persistence.save("main", msgs),
)

12.2 Agent 问题

Q: Agent 无法调用 send_message

确保通信流已正确定义：

agency = Agency(
    ceo,
    communication_flows=[ceo > developer],  # 必须定义
)

# 增加超时时间

    name="SlowAgent",
    model_settings=ModelSettings(
        timeout=120,  # 2 分钟
    ),
)

12.3 工具问题

Q: 工具参数验证失败

确保参数类型正确：

@function_tool
def my_tool(count: int, name: str) -> str:
    """工具描述。"""
    return f"处理 {count} 个 {name}"

Q: 工具执行错误

# 添加错误处理
@function_tool
def risky_tool(data: str) -> str:
    """工具描述。"""
    try:
        return process(data)
    except Exception as e:
        return f"错误: {str(e)}"
