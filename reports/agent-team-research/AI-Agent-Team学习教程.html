一、快速开始

1.1 什么是 Agent Team？

Agent Team 是一组协作完成任务的 AI 智能体。就像人类团队一样，每个 Agent 有自己的角色和职责：

人类团队                    AI Agent Team
┌─────────┐                ┌─────────┐
│ 产品经理 │                │ PM Agent │
├─────────┤                ├─────────┤
│ 架构师   │       →        │ Architect│
├─────────┤                ├─────────┤
│ 开发者   │                │ Engineer│
├─────────┤                ├─────────┤
│ 测试工程师│               │ QA Agent │
└─────────┘                └─────────┘

1.2 框架选择指南

你的需求 推荐框架 原因
新手入门 CrewAI API 简洁，概念清晰
软件开发 MetaGPT 内置 SDLC 流程
研究分析 AutoGen 代码执行能力强
复杂流程 LangGraph 图结构灵活可控

1.3 5 分钟上手 CrewAI

# 安装
pip install crewai

# 创建脚本
cat > my_crew.py << 'EOF'
from crewai import Agent, Task, Crew

# 定义 Agent
researcher = Agent(
    role="研究员",
    goal="研究 AI 趋势",
    backstory="你是一位资深的 AI 研究员"
)

writer = Agent(
    role="撰稿人",
    goal="撰写技术文章",
    backstory="你是一位技术作家"
)

# 定义任务
research_task = Task(
    description="研究 2024 年 AI 发展趋势",
    agent=researcher
)

write_task = Task(
    description="基于研究结果撰写一篇文章",
    agent=writer
)

# 组建团队并执行
crew = Crew(
    agents=[researcher, writer],
    tasks=[research_task, write_task]
)

result = crew.kickoff()
print(result)
EOF

# 运行
python my_crew.py

2.4 Task 详解

from crewai import Task

research_task = Task(
    # 任务描述
    description="""
    研究 {topic} 领域的最新发展。
    关注以下方面：
    1. 技术趋势
    2. 主要参与者
    3. 未来预测
    """,

    # 期望输出
    expected_output="""
    一份结构化的研究报告，包含：
    - 执行摘要
    - 关键发现
    - 详细分析
    - 建议
    """,

    # 指派 Agent
    agent=researcher,

    # 输出配置
    output_file="research_report.md",

    # 异步执行
    async_execution=False,

    # 回调函数
    callback=lambda output: print(f"任务完成: {output}"),
)

2.5 Crew 组装与执行

from crewai import Crew, Process

# Sequential 流程
crew = Crew(
    agents=[researcher, writer, editor],
    tasks=[research_task, write_task, edit_task],
    process=Process.sequential,
    verbose=True,
    memory=True,  # 团队记忆
)

# 执行
result = crew.kickoff()

# 带输入执行
result = crew.kickoff(inputs={"topic": "大语言模型"})

# Hierarchical 流程（需要 Manager）
crew = Crew(
    agents=[researcher, writer, editor],
    tasks=[main_task],
    process=Process.hierarchical,
    manager_llm=llm,  # Manager 使用的 LLM
    verbose=True,
)

@tool("计算器工具")
def calculator(expression: str) -> str:
    """执行数学计算"""
    try:
        return str(eval(expression))
    except Exception as e:
        return f"计算错误: {e}"

agent = Agent(
    role="数据分析师",
    tools=[calculator],
)

2.7 完整示例：内容创作团队

from crewai import Agent, Task, Crew, Process
from crewai_tools import SerperDevTool, ScrapeWebsiteTool
from langchain_openai import ChatOpenAI

# 配置
llm = ChatOpenAI(model="gpt-4")
search_tool = SerperDevTool()
scrape_tool = ScrapeWebsiteTool()

# Agent 1: 研究员
researcher = Agent(
    role="内容研究员",
    goal="研究主题并收集关键信息",
    backstory="你是专业的内容研究员，擅长发现有价值的信息",
    tools=[search_tool, scrape_tool],
    llm=llm,
    verbose=True,
)

# Agent 2: 撰稿人
writer = Agent(
    role="技术撰稿人",
    goal="撰写高质量的技术文章",
    backstory="你是资深技术作家，擅长将复杂概念简单化",
    llm=llm,
    verbose=True,
)

# Agent 3: 编辑
editor = Agent(
    role="内容编辑",
    goal="审核和优化文章质量",
    backstory="你是严谨的编辑，确保内容准确和流畅",
    llm=llm,
    verbose=True,
)

# Task 1: 研究
research_task = Task(
    description="研究 {topic} 的核心概念和最新发展",
    expected_output="包含关键点的结构化研究报告",
    agent=researcher,
)

# Task 2: 撰写
write_task = Task(
    description="基于研究结果撰写一篇关于 {topic} 的文章",
    expected_output="结构清晰的技术文章（1000字左右）",
    agent=writer,
    context=[research_task],  # 依赖研究任务
)

# Task 3: 编辑
edit_task = Task(
    description="审核并优化文章",
    expected_output="最终版本的文章",
    agent=editor,
    context=[write_task],
    output_file="final_article.md",
)

# 组建团队
crew = Crew(
    agents=[researcher, writer, editor],
    tasks=[research_task, write_task, edit_task],
    process=Process.sequential,
    verbose=True,
)

# 执行
result = crew.kickoff(inputs={"topic": "多智能体系统"})
print(result)

3.5 群聊模式

import autogen

config_list = [{"model": "gpt-4", "api_key": "your-api-key"}]
llm_config = {"config_list": config_list}

# 创建多个 Agent
researcher = autogen.AssistantAgent(
    name="researcher",
    llm_config=llm_config,
    system_message="你是研究员，负责收集和分析信息。",
)

coder = autogen.AssistantAgent(
    name="coder",
    llm_config=llm_config,
    system_message="你是程序员，负责编写代码。",
)

reviewer = autogen.AssistantAgent(
    name="reviewer",
    llm_config=llm_config,
    system_message="你是代码审核员，负责检查代码质量。",
)

# 用户代理
user = autogen.UserProxyAgent(
    name="user",
    human_input_mode="NEVER",
    code_execution_config={"work_dir": "workspace"},
)

# 创建群聊
groupchat = autogen.GroupChat(
    agents=[user, researcher, coder, reviewer],
    messages=[],
    max_round=12,  # 最大轮数
)

# 创建管理器
manager = autogen.GroupChatManager(
    groupchat=groupchat,
    llm_config=llm_config,
)

# 发起群聊
user.initiate_chat(
    manager,
    message="分析一个 CSV 文件并生成可视化报告",
)

3.6 Human-in-the-Loop

import autogen

# ALWAYS: 每次都请求人类输入
user_always = autogen.UserProxyAgent(
    name="user_always",
    human_input_mode="ALWAYS",
)

# NEVER: 完全自动
user_never = autogen.UserProxyAgent(
    name="user_never",
    human_input_mode="NEVER",
)

# TERMINATE: 仅在终止时请求
user_terminate = autogen.UserProxyAgent(
    name="user_terminate",
    human_input_mode="TERMINATE",
    is_termination_msg=lambda x: x.get("content", "").rstrip().endswith("TERMINATE"),
)

4.4 自定义角色

from metagpt.roles import Role
from metagpt.actions import Action

# 定义自定义动作
class WriteDocumentation(Action):
    """编写文档"""

    async def run(self, code_content: str):
        # 使用 LLM 生成文档
        prompt = f"为以下代码编写文档:\n{code_content}"
        return await self.llm.aask(prompt)

# 定义自定义角色
class TechnicalWriter(Role):
    """技术文档作者"""

    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self.set_actions([WriteDocumentation])

    async def _act(self) -> str:
        # 执行动作
        todo = self.rc.todo
        result = await todo.run(self.rc.history[-1].content)
        return result

# 使用自定义角色
writer = TechnicalWriter(name="TechWriter", profile="技术文档作者")

5.3 基础图构建

from typing import TypedDict, Annotated
from langgraph.graph import StateGraph, END
import operator

# 定义状态
class State(TypedDict):
    messages: Annotated[list, operator.add]
    next: str

# 定义节点
def node_a(state: State) -> dict:
    return {"messages": ["A 处理完成"], "next": "b"}

def node_b(state: State) -> dict:
    return {"messages": ["B 处理完成"], "next": "end"}

# 构建图
graph = StateGraph(State)

# 添加节点
graph.add_node("a", node_a)
graph.add_node("b", node_b)

# 设置入口
graph.set_entry_point("a")

# 添加边
graph.add_edge("a", "b")
graph.add_edge("b", END)

# 编译
app = graph.compile()

# 执行
result = app.invoke({"messages": [], "next": ""})
print(result)

5.4 条件分支

from typing import Literal
from langgraph.graph import StateGraph, END

# 定义状态
class AgentState(TypedDict):
    input: str
    output: str
    decision: str

# 路由函数
def router(state: AgentState) -> Literal["research", "code", "end"]:
    if "研究" in state["input"]:
        return "research"
    elif "代码" in state["input"]:
        return "code"
    else:
        return "end"

# 节点函数
def research_node(state: AgentState) -> dict:
    return {"output": "研究结果...", "decision": "done"}

def code_node(state: AgentState) -> dict:
    return {"output": "代码实现...", "decision": "done"}

# 构建图
graph = StateGraph(AgentState)

graph.add_node("research", research_node)
graph.add_node("code", code_node)

graph.set_entry_point("router")

# 条件边
graph.add_conditional_edges(
    "router",
    router,
    {
        "research": "research",
        "code": "code",
        "end": END,
    }
)

graph.add_edge("research", END)
graph.add_edge("code", END)

5.5 循环结构

from typing import TypedDict, Annotated
from langgraph.graph import StateGraph, END
import operator

class LoopState(TypedDict):
    count: Annotated[int, operator.add]
    max_count: int

def increment(state: LoopState) -> dict:
    return {"count": 1}

def should_continue(state: LoopState) -> str:
    if state["count"] >= state["max_count"]:
        return "end"
    return "continue"

# 构建循环图
graph = StateGraph(LoopState)

graph.add_node("increment", increment)

graph.set_entry_point("increment")

graph.add_conditional_edges(
    "increment",
    should_continue,
    {"continue": "increment", "end": END}
)

app = graph.compile()

# 执行
result = app.invoke({"count": 0, "max_count": 5})
print(result)  # count: 5

5.6 多 Agent 协作

from typing import TypedDict, Annotated, Sequence
from langgraph.graph import StateGraph, END
from langchain_core.messages import BaseMessage, HumanMessage, AIMessage
from langchain_openai import ChatOpenAI
import operator

# 状态定义
class TeamState(TypedDict):
    messages: Annotated[Sequence[BaseMessage], operator.add]
    next_agent: str
    iterations: Annotated[int, operator.add]

# LLM
llm = ChatOpenAI(model="gpt-4")

# Agent 节点
def researcher_node(state: TeamState) -> dict:
    response = llm.invoke([
        {"role": "system", "content": "你是研究员，专注于收集信息。"},
        *state["messages"]
    ])
    return {
        "messages": [response],
        "next_agent": "writer"
    }

def writer_node(state: TeamState) -> dict:
    response = llm.invoke([
        {"role": "system", "content": "你是作家，专注于内容创作。"},
        *state["messages"]
    ])
    return {
        "messages": [response],
        "iterations": 1
    }

def router(state: TeamState) -> str:
    if state["iterations"] >= 3:
        return "end"
    if state["next_agent"] == "writer":
        return "writer"
    return "researcher"

# 构建团队图
workflow = StateGraph(TeamState)

workflow.add_node("researcher", researcher_node)
workflow.add_node("writer", writer_node)

workflow.set_entry_point("researcher")

workflow.add_edge("researcher", "writer")
workflow.add_conditional_edges(
    "writer",
    router,
    {"researcher": "researcher", "end": END}
)

app = workflow.compile()

# 执行
result = app.invoke({
    "messages": [HumanMessage(content="写一篇关于 AI Agent 的文章")],
    "next_agent": "researcher",
    "iterations": 0
})

七、最佳实践

7.1 Agent 设计原则

  1. 单一职责：每个 Agent 只做一件事
  2. 清晰边界：明确 Agent 的输入和输出
  3. 适当粒度：不要过度细分或过于笼统
  4. 工具赋能：给 Agent 提供必要的工具

7.2 任务拆分技巧

好任务：
├── 具体明确
├── 可验证输出
├── 依赖关系清晰
└── 粒度适中

坏任务：
├── 模糊不清
├── 无法验证
├── 循环依赖
└── 粒度过大或过小

7.3 成本控制

  1. 选择合适的模型：简单任务用小模型
  2. 缓存结果：避免重复调用
  3. 限制迭代：设置最大迭代次数
  4. 并行执行：利用并行减少时间

八、常见问题

Q1: 如何选择框架？

  • 快速原型 → CrewAI
  • 软件开发 → MetaGPT
  • 数据分析 → AutoGen
  • 复杂流程 → LangGraph

Q2: Agent 执行卡住怎么办？

  1. 检查 LLM 配置
  2. 设置最大迭代次数
  3. 添加超时机制
  4. 检查工具是否正常

Q3: 如何调试多 Agent 系统？

  1. 启用 verbose 模式
  2. 查看消息历史
  3. 检查状态流转
  4. 使用可视化工具

Q4: 成本太高怎么办？

  1. 使用更小的模型
  2. 缓存中间结果
  3. 减少迭代次数
  4. 合并相似任务
